# 1. Import libraries
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf

from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_squared_error
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Input, SimpleRNN, LSTM, GRU, Dense

# 2. Load Airline Passengers dataset
url = "https://raw.githubusercontent.com/jbrownlee/Datasets/master/airline-passengers.csv"
df = pd.read_csv(url)

print(df.head())
print( df.shape)

data = df["Passengers"].values.reshape(-1, 1)

# 3. Normalize data between 0 and 1
scaler = MinMaxScaler()
scaled = scaler.fit_transform(data)

# 4. Create sequences (previous 12 months -> next month)
LOOK_BACK = 12
X, y = [], []

for i in range(len(scaled) - LOOK_BACK):
    X.append(scaled[i:i + LOOK_BACK])
    y.append(scaled[i + LOOK_BACK])

X = np.array(X)
y = np.array(y)

# 5. Split chronologically: 80% train, 20% test
split = int(len(X) * 0.8)

X_train, X_test = X[:split], X[split:]
y_train, y_test = y[:split], y[split:]

print("X shape:", X.shape)
print("Training samples:", len(X_train))
print("Testing samples:", len(X_test))

# 6. Train RNN, LSTM and GRU
layers = {
    "RNN": SimpleRNN,
    "LSTM": LSTM,
    "GRU": GRU
}

results = {}

for name, layer in layers.items():
    model = Sequential([
        Input(shape=(LOOK_BACK, 1)),
        layer(50, activation="tanh"),
        Dense(1)
    ])

    model.compile(optimizer="adam", loss="mse")

    model.fit(
        X_train, y_train,
        epochs=20,
        batch_size=8,
        verbose=0
    )

    # 7. Predict and convert to original scale
    pred = model.predict(X_test, verbose=0)

    actual = scaler.inverse_transform(y_test)
    predicted = scaler.inverse_transform(pred)

    rmse = np.sqrt(mean_squared_error(actual, predicted))
    results[name] = (predicted, rmse)

    print(name, "RMSE:", round(rmse, 2))

# 8. Plot actual and predicted values
plt.figure(figsize=(10, 5))
plt.plot(actual, label="Actual")

for name, (predicted, rmse) in results.items():
    plt.plot(predicted, label=name)

plt.title("Airline Passenger Forecasting")
plt.xlabel("Test Time Steps")
plt.ylabel("Number of Passengers")
plt.legend()
plt.show()


# 9. Find and print the best model
best_model = min(results, key=lambda name: results[name][1])

print("\nBest Model:", best_model)
print("Best RMSE:", round(results[best_model][1], 2))
